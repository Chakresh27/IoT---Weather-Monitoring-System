char auth[] = "LesE00ClovyiU8_4KzXdnEpsSJHjjmw2"; //Enter the Auth code which was send by Blink
char ssid[] = "A50";  //Enter your WIFI Name
char pass[] = "12345678";  //Enter your WIFI Password

DHT dht(D3, DHT11); //(sensor pin,sensor type) // dht12 in case of trouble shooting
BlynkTimer timer;

void weather() {
  float h = dht.readHumidity();
  float t = dht.readTemperature();
  float r = analogRead(A0);
  bool l = digitalRead(D4); //option al

  r = map(r, 0, 1023, 600, 0);
  if (isnan(h) || isnan(t)) {
    Serial.println("Failed to read from DHT sensor!");
    //return;
  }

  Blynk.virtualWrite(V0, t);  //V0 is for Temperature
  Blynk.virtualWrite(V1, h);  //V1 is for Humidity
  Blynk.virtualWrite(V2, r);  //V2 is for Rainfall

  if (l == 1) {
    WidgetLED led1(V3);
    led1.on();
  } else if (l == 0) {
    WidgetLED led1(V3);
    led1.off();
  }
}

void setup() {
  Serial.begin(9600); 
  Blynk.begin(auth, ssid, pass);
  dht.begin();
  timer.setInterval(10L, weather);
}

void loop() {
  Blynk.run(); // Initiates Blynk
  timer.run(); // Initiates SimpleTimer
}
